  <button id="drawBtn" onclick="drawQuestion()">Losuj pytanie</button>

<script>
const TOTAL = 365;

const questions = [
"Jaka mała rzecz dziś poprawiła Ci humor?",
"Co najbardziej lubisz robić rano?",
"Jaki zapach kojarzy Ci się z czymś miłym?",
"Jakie jedzenie zawsze poprawia Ci nastrój?",
"Co było dziś najmilszym momentem?",
"Jaką piosenkę mógłbyś/mogłabyś słuchać w kółko?",
"Jakie miejsce daje Ci spokój?",
"Co lubisz robić bez pośpiechu?",
"Jaki napój najbardziej kojarzy Ci się z relaksem?",
"Co ostatnio Cię rozbawiło?"
  // ← TU DOKLEJ CAŁĄ PULĘ 365 PYTAŃ
];

let unused = JSON.parse(localStorage.getItem("unusedQuestions"));
let lastDate = localStorage.getItem("lastDrawDate");
const today = new Date().toDateString();

if(!unused || unused.length===0){
  unused=[...questions];
  localStorage.setItem("unusedQuestions",JSON.stringify(unused));
}

const usedCount = TOTAL - unused.length;
document.getElementById("counter").innerText =
  `Dzień ${usedCount + 1} / ${TOTAL}`;

if(lastDate===today){
  document.getElementById("drawBtn").disabled=true;
  document.getElementById("info").innerText="Wróć jutro po kolejne pytanie 💛";
}

function drawQuestion(){
  if(unused.length===0){
    document.getElementById("question").innerText="To było ostatnie pytanie 💛";
    document.getElementById("drawBtn").disabled=true;
    return;
  }

  const index=Math.floor(Math.random()*unused.length);
  const q=unused.splice(index,1)[0];

  localStorage.setItem("unusedQuestions",JSON.stringify(unused));
  localStorage.setItem("lastDrawDate",today);

  document.getElementById("question").innerText=q;
  document.getElementById("drawBtn").disabled=true;
  document.getElementById("info").innerText="Wróć jutro po kolejne pytanie 💛";

  const newUsed = TOTAL - unused.length;
  document.getElementById("counter").innerText =
    `Dzień ${newUsed} / ${TOTAL}`;
}
</script>
